    // Validate slug function
    function validateSlug(slug) {
        // Check if slug ends with underscore
        if (slug.endsWith('_')) {
            return { valid: false, reason: 'Cannot end with underscore (_)' };
        }
        
        // Check for special characters (allow only alphanumeric and underscore)
        const invalidChars = slug.match(/[^a-zA-Z0-9_]/g);
        if (invalidChars) {
            return { valid: false, reason: `Contains invalid character(s): ${[...new Set(invalidChars)].join(', ')}` };
        }
        
        return { valid: true };
    }

    // Create one link
    async function createLink(url, domain, slug) {
        console.log(`\n→ Creating: ${slug}`);

        const btn = document.querySelector('[data-dialog="logger-create"]');
        if (!btn) {
            console.log('❌ Create button not found');
            return false;
        }
        btn.click();
        await new Promise(r => setTimeout(r, 80));

        const urlInput = document.querySelector('input[name="destination"]');
        if (urlInput) {
            urlInput.value = url;
            urlInput.dispatchEvent(new Event('input', { bubbles: true }));
        }
        await new Promise(r => setTimeout(r, 30));

        const domainSelect = document.querySelector('.domain-select-current');
        if (domainSelect) {
            domainSelect.click();
            await new Promise(r => setTimeout(r, 30));
            const domainOption = document.querySelector(`.domain-select-item[data-domain="${domain}"]`);
            if (domainOption) {
                domainOption.click();
                await new Promise(r => setTimeout(r, 25));
            }
        }

        const slugInput = document.querySelector('input[name="manual"]');
        if (slugInput) {
            slugInput.value = slug;
            slugInput.dispatchEvent(new Event('input', { bubbles: true }));
        }
        await new Promise(r => setTimeout(r, 30));

        const submitBtn = document.querySelector('#logger-create button[type="submit"]');
        if (!submitBtn) {
            console.log('❌ Submit button not found');
            return false;
        }
        submitBtn.click();
        console.log('✓ Submit clicked!');

        await new Promise(r => setTimeout(r, 50));

        console.log(`✅ Link created: https://${domain}/${slug}`);

        return true;
    }

    // Main process
    async function run() {
        console.log('\n🚀 RUN() FUNCTION CALLED');

        const state = JSON.parse(sessionStorage.getItem('batch_state') || 'null');

        if (!state || !state.running) {
            console.log('⏸️ Batch is not running. Click START to begin.');
            return;
        }

        const remainingSlugs = state.remainingSlugs || [];
        const totalSlugs = state.totalSlugs || remainingSlugs.length;
        const completed = totalSlugs - remainingSlugs.length;

        console.log(`\n📊 Current State:`);
        console.log(`   Completed: ${completed}/${totalSlugs}`);
        console.log(`   Remaining: ${remainingSlugs.length}`);
        console.log(`   Success: ${state.success} | Failed: ${state.fail}`);
        console.log(`   Next slug: ${remainingSlugs[0] || 'DONE'}`);

        if (remainingSlugs.length === 0) {
            console.log('\n╔════════════════════════════════════╗');
            console.log('║  🎉 ALL LINKS CREATED!             ║');
            console.log(`║  Success: ${state.success}                        ║`);
            console.log(`║  Failed: ${state.fail}                         ║`);
            console.log('╚════════════════════════════════════╝');

            // Remove background overlay when complete
            removeBackgroundOverlay();

            // Generate all links from original slugs
            const allLinks = state.originalSlugs.map(slug => `https://${CONFIG.domain}/${slug}`).join('\n');

            // Mark batch as complete but keep the data
            state.running = false;
            state.completed = true;
            sessionStorage.setItem('batch_state', JSON.stringify(state));

            navigator.clipboard.writeText(allLinks).then(() => {
                console.log('📋 All links copied to clipboard!');
                alert(`✅ Batch Complete!\n\n${state.success} links created!\n\nAll ${state.originalSlugs.length} links have been copied to your clipboard!\n\nYou can copy them again using the COPY ALL LINKS button.`);
            }).catch(err => {
                console.log('❌ Failed to copy to clipboard:', err);
                alert(`✅ Batch Complete!\n\n${state.success} links created!\n\nLinks:\n${allLinks}\n\nYou can copy them again using the COPY ALL LINKS button.`);
            });

            updateControlPanel();
            return;
        }

        const slug = remainingSlugs[0];
        console.log(`\n[${completed + 1}/${totalSlugs}] 🔗 Creating: ${slug}`);

        const success = await createLink(state.targetUrl, CONFIG.domain, slug);

        console.log(`\n🗑️ REMOVING slug "${slug}" from queue...`);
        remainingSlugs.shift();

        if (success) state.success++;
        else state.fail++;

        state.remainingSlugs = remainingSlugs;
        sessionStorage.setItem('batch_state', JSON.stringify(state));

        console.log(`✅ State updated and saved!`);
        console.log(`   Removed: "${slug}"`);
        console.log(`   Remaining: ${remainingSlugs.length}`);
        console.log(`   Next: ${remainingSlugs[0] || 'DONE'}`);

        console.log('\n🔄 Returning to account page...');

        setTimeout(() => {
            window.location.href = 'https://iplogger.org/account/';
        }, 500);
    }

    // Validate slugs in real-time
    function validateSlugsInput() {
        const textarea = document.getElementById('slugs-input');
        const errorContainer = document.getElementById('validation-errors');
        const startBtn = document.getElementById('modal-start-btn');
        
        if (!textarea || !errorContainer || !startBtn) return;

        const slugsText = textarea.value.trim();
        
        if (!slugsText) {
            errorContainer.innerHTML = '';
            startBtn.disabled = false;
            startBtn.style.opacity = '1';
            startBtn.style.cursor = 'pointer';
            return;
        }

        const slugs = slugsText.split('\n')
            .map((s, idx) => ({ slug: s.trim(), line: idx + 1 }))
            .filter(item => item.slug.length > 0);

        const invalidSlugs = [];
        
        slugs.forEach(item => {
            const validation = validateSlug(item.slug);
            if (!validation.valid) {
                invalidSlugs.push({
                    line: item.line,
                    slug: item.slug,
                    reason: validation.reason
                });
            }
        });

        if (invalidSlugs.length > 0) {
            errorContainer.innerHTML = `
                <div style="
                    background: #fee2e2;
                    border: 2px solid #ef4444;
                    border-radius: 6px;
                    padding: 12px;
                    margin-top: 10px;
                ">
                    <div style="font-weight: bold; color: #dc2626; margin-bottom: 8px;">
                        ❌ Invalid Slugs Found (${invalidSlugs.length}):
                    </div>
                    <div style="font-size: 12px; color: #991b1b; max-height: 150px; overflow-y: auto;">
                        ${invalidSlugs.map(item => `
                            <div style="margin-bottom: 6px; padding: 6px; background: white; border-radius: 4px;">
                                <strong>Line ${item.line}:</strong> <code style="background: #fecaca; padding: 2px 4px; border-radius: 3px;">${item.slug}</code><br>
                                <span style="color: #dc2626;">→ ${item.reason}</span>
                            </div>
                        `).join('')}
                    </div>
                    <div style="font-size: 11px; color: #991b1b; margin-top: 8px; font-style: italic;">
                        ℹ️ Only letters, numbers, and underscores (_) are allowed. Underscores cannot be at the end.
                    </div>
                </div>
            `;
            
            startBtn.disabled = true;
            startBtn.style.opacity = '0.5';
            startBtn.style.cursor = 'not-allowed';
        } else {
            errorContainer.innerHTML = `
                <div style="
                    background: #d1fae5;
                    border: 2px solid #10b981;
                    border-radius: 6px;
                    padding: 8px 12px;
                    margin-top: 10px;
                    color: #065f46;
                    font-size: 13px;
                    font-weight: bold;
                ">
                    ✅ All ${slugs.length} slug${slugs.length !== 1 ? 's' : ''} valid!
                </div>
            `;
            
            startBtn.disabled = false;
            startBtn.style.opacity = '1';
            startBtn.style.cursor = 'pointer';
        }
    }

    // Start batch
    function startBatch(url, slugs) {
        const state = {
            running: true,
            targetUrl: url,
            remainingSlugs: [...slugs],
            originalSlugs: [...slugs],
            totalSlugs: slugs.length,
            success: 0,
            fail: 0
        };
        sessionStorage.setItem('batch_state', JSON.stringify(state));
        console.log('▶️ Batch STARTED!');
        console.log(`   Target URL: ${url}`);
        console.log(`   Total slugs: ${slugs.length}`);

        // Create background overlay when batch starts
        createBackgroundOverlay();

        updateControlPanel();
        setTimeout(() => run(), 50);
    }

    // Stop batch
    function stopBatch() {
        const state = JSON.parse(sessionStorage.getItem('batch_state') || 'null');

        // Remove background overlay when stopping
        removeBackgroundOverlay();

        if (state && state.originalSlugs && state.originalSlugs.length > 0) {
            const copy = confirm(`You have ${state.originalSlugs.length} slugs.\n\nDo you want to copy all the shortened links before stopping?`);
            if (copy) {
                copyAllLinks();
            }
        }

        sessionStorage.removeItem('batch_state');
        console.log('⏸️ Batch STOPPED!');
        updateControlPanel();
    }

    // Reset batch
    function resetBatch() {
        removeBackgroundOverlay();
        sessionStorage.removeItem('batch_state');
        console.log('🔄 Batch RESET!');
        location.reload();
    }

    // Check if we should continue batch
    function checkAndContinueBatch() {
        console.log('\n🔍 Checking if batch should continue...');

        const state = JSON.parse(sessionStorage.getItem('batch_state') || 'null');

        if (!state) {
            console.log('   No state found');
            return;
        }

        console.log(`   State found: running=${state.running}, completed=${state.completed}`);
        console.log(`   Remaining slugs: ${state.remainingSlugs?.length || 0}`);
        console.log(`   Original slugs: ${state.originalSlugs?.length || 0}`);

        if (state.completed) {
            console.log('   ✅ Batch is complete! Showing completion panel.');
            updateControlPanel();
            return;
        }

        if (state && state.running && state.remainingSlugs && state.remainingSlugs.length > 0) {
            console.log('   ✅ Batch should continue!');
            console.log(`   Next slug: ${state.remainingSlugs[0]}`);
            console.log('   Starting in 1.5 seconds...\n');

            // Show background overlay if batch is running
            createBackgroundOverlay();

            updateControlPanel();

            setTimeout(() => {
                console.log('⏰ Time passed, calling run()...');
                run();
            }, 1500);
        } else {
            console.log('   ⏸️ Batch not running or no slugs left');
            updateControlPanel();
        }
